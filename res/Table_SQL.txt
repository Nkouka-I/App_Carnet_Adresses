CREATE TABLE Contact (
id INT PRIMARY KEY,
nom VARCHAR(254) NOT NULL,
email VARCHAR(254) NOT NULL
);

CREATE TABLE Etiquette (
id INT PRIMARY KEY,
libelle VARCHAR(100) NOT NULL UNIQUE
);

-- Table entre Contact et Étiquette
CREATE TABLE Contact_etiquette (
contact_id INT,
etiquette_id INT,
PRIMARY KEY (contact_id, etiquette_id),
FOREIGN KEY (contact_id) REFERENCES Contact(id) ON DELETE CASCADE,
FOREIGN KEY (etiquette_id) REFERENCES Etiquette(id) ON DELETE CASCADE
);

CREATE TABLE ListeDiffusion (
id INT PRIMARY KEY,
nom VARCHAR(50) NOT NULL UNIQUE
);

-- 1 critère = 1 liste + 1 étiquette + un booléen inclus/exclu)
CREATE TABLE Critere (
id INT PRIMARY KEY,
liste_id INT NOT NULL,
etiquette_id INT NOT NULL,
inclus BOOLEAN NOT NULL, -- true = il doit l'avoir, false = il ne doit pas l'avoir
FOREIGN KEY (liste_id) REFERENCES ListeDiffusion(id) ON DELETE CASCADE,
FOREIGN KEY (etiquette_id) REFERENCES Etiquette(id) ON DELETE CASCADE
);
